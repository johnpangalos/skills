package middleware

import (
	"bufio"
	"io"
	"net"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"
)

func rawGet(t *testing.T, h http.Handler) string {
	srv := httptest.NewServer(h)
	defer srv.Close()
	c, err := net.Dial("tcp", srv.Listener.Addr().String())
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close()
	io.WriteString(c, "GET / HTTP/1.1\r\nHost: x\r\nConnection: close\r\n\r\n")
	b, _ := io.ReadAll(c)
	return strings.ReplaceAll(string(b), "\r\n", "|")
}

func TestZZHijackAfterWrite(t *testing.T) {
	h := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Write([]byte("hi"))
		conn, rw, err := http.NewResponseController(w).Hijack()
		if err != nil {
			t.Log("hijack err", err)
			return
		}
		rw.WriteString("RAW")
		rw.Flush()
		_, werr := w.Write([]byte("after"))
		t.Logf("write after hijack err=%v", werr)
		conn.Close()
	})
	t.Logf("plain: %q", rawGet(t, h))
	t.Logf("etag:  %q", rawGet(t, ETag(h)))
}

func TestZZLateTrailer(t *testing.T) {
	h := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Trailer", "X-Sum")
		w.Write([]byte("abc"))
		w.Header().Set("X-Sum", "zz")
	})
	for name, hh := range map[string]http.Handler{"plain": h, "etag": ETag(h)} {
		srv := httptest.NewServer(hh)
		resp, _ := http.Get(srv.URL)
		io.ReadAll(resp.Body)
		t.Logf("%s header=%q trailer=%q", name, resp.Header.Get("X-Sum"), resp.Trailer.Get("X-Sum"))
		srv.Close()
	}
}

func TestZZ1xxAfter200(t *testing.T) {
	h := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(200)
		w.Header().Set("Link", "</a>; rel=preload")
		w.WriteHeader(103)
		w.Write([]byte("x"))
	})
	_ = bufio.NewReader
	t.Logf("plain: %q", rawGet(t, h))
	t.Logf("etag:  %q", rawGet(t, ETag(h)))
}
